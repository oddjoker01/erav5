moe_pipeline.py · MoE + MoEBlockT1
class MoE(nn.Module):
    def __init__(self, c, E=4, k=2):
        super().__init__()
        self.E, self.k = E, k
        self.router = nn.Linear(c.D, E, bias=False)                  # D → E scores
        self.experts = nn.ModuleList([SwiGLU(c) for _ in range(E)])  # 4 × dense FFN
        self.aux = None     # balancing loss (C6)
        self.load = None    # picks per expert (logs)

class MoEBlock(nn.Module):
    def __init__(self, c, E=4, k=2):
        super().__init__()
        self.attn = Attention(c)          # unchanged
        self.n1 = RMSNorm(c.D)
        self.n2 = RMSNorm(c.D)
        self.ffn = MoE(c, E, k)           # was: SwiGLU(c)

moe_pipeline.py · MoE.forwardL35–39
xf = x.reshape(-1, D)                         # every token = 1 row
probs = self.router(xf).float().softmax(-1)   # fp32 softmax, (N, E)
topw, topi = probs.topk(self.k, dim=-1)       # weights + ids, (N, k)
topw = topw / topw.sum(-1, keepdim=True)      # rescale → sum 1

moe_pipeline.py · end of MoE.forwardL52–55
f = F.one_hot(topi, self.E).float().sum((0, 1)) / (xf.size(0) * self.k)
#   topi (N,k) → one-hot (N,k,E) → count per expert → ÷ all picks
P = probs.mean(0)                       # mean prob per expert (has grad)
self.aux = self.E * (f * P).sum()       # 1.0 when balanced
self.load = f.detach()                  # logged as load_min / load_max
moe_pipeline.py · helper
def aux_loss(model):                    # mean over the 7 MoE layers
    layers = moe_layers(model)
    return sum(m.aux for m in layers) / len(layers)

moe_pipeline.py · param_counts
def param_counts(model, E, k):
    total = sum(p.numel() for p in model.parameters())
    expert = sum(p.numel() for m in moe_layers(model) for p in m.experts.parameters())
    active = total - expert + expert * k // E    # only k of E experts run
    return total, active
# 56.86M − 49.55M + 49.55M × 2/4 = 32.09M active
